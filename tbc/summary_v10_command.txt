        else:

            if not report_date:

                    latest_data = latest.json()

            if report_date:

                if not valid_date:
                    bot.sendMessage("Invalid date. Use YYYY-MM-DD.")
                else:
                    url = "https://raw.githubusercontent.com/fyroz118/Legion-OSINT/main/docs/feed/archive/" + report_date + "/intelligence_report.json"
                    response = HTTP.get(url)

                    if response.status_code != 200:
                        bot.sendMessage("EXPANDED INTELLIGENCE REPORT NOT FOUND for " + report_date + ". No compact-summary fallback is used.")
                    else:
                        report = response.json()

                        if not report.get("executive_assessment") or not report.get("report_sections"):
                            bot.sendMessage("Archive is not the expanded intelligence report.")
                        else:
                            executive = report.get("executive_assessment", {})
                            lines = [
                                "LEGION-OSINT DAILY INTELLIGENCE SUMMARY — V10",
                                "Reporting date: " + str(report.get("report_date", report_date)) + " | Timezone: " + str(report.get("timezone", "Asia/Dhaka")),
                                "Generated: " + str(report.get("generated_at", "unknown")) + " (timestamp as stored by collector)",
                                "Classification: UNVERIFIED SOURCE-LEAD ASSESSMENT",

                                "EXECUTIVE COLLECTION AND RELEVANCE PICTURE",

                                "Relevant source posts: " + str(executive.get("relevant_source_posts", 0)),

                                "Upstream candidate source posts: " + str(executive.get("incident_candidate_source_posts", 0)),
                                "Exact-text candidate groups: " + str(executive.get("incident_candidate_exact_text_groups", 0)),
                                "Bangladesh total source posts: " + str(executive.get("bangladesh_total_source_posts", 0)),
                                "Bangladesh relevant source posts: " + str(executive.get("bangladesh_relevant_source_posts", 0)),
                                "Telegram: " + str(executive.get("telegram_posts", 0)) + " | RSS/Atom: " + str(executive.get("rss_articles", 0)),
                                "Regions represented: " + str(executive.get("regions_represented", 0)) + " | Regions with relevant leads: " + str(executive.get("regions_with_relevant_leads", 0)),
                                "Regional counts reconcile: " + str(executive.get("regional_counts_reconcile_to_archive", "not reported")) + " | Regional total: " + str(executive.get("regional_source_post_total_check", "not reported")),
                                str(executive.get("key_caveat", "")),
                                "",
                                "V10 REVIEW RULE: physical incident, civil unrest, security action, threat/preparation and political commentary are separated. Same-source exact-text repeats are collapsed in the display only; no incident is verified.",
                                ""
                            ]

                            for section in report.get("report_sections", []):
                                heading = str(section.get("heading", ""))
                                lines.append(heading)
                                lines.append(str(section.get("text", "")))

                                if section.get("metrics"):
                                    metrics = section.get("metrics", {})
                                    lines.append("Collected: " + str(metrics.get("collected_posts", 0)) + " | Relevant: " + str(metrics.get("relevant_source_posts", 0)) + " | Excluded: " + str(metrics.get("excluded_low_relevance_posts", 0)))

                                if section.get("bangladesh_total_posts") is not None:
                                    lines.append("Bangladesh total: " + str(section.get("bangladesh_total_posts", 0)) + " | Relevant: " + str(section.get("bangladesh_relevant_posts", 0)))
                                    lines.append("Category counts: " + str(section.get("category_counts", {})))
                                    lines.append("District-name mentions (not geolocation): " + str(section.get("district_name_mentions", {})))

                                if section.get("regions"):
                                    for region in section.get("regions", []):
                                        lines.append("- " + str(region.get("region", "Unknown")) + ": " + str(region.get("source_post_count", 0)) + " total; " + str(region.get("relevant_source_post_count", 0)) + " relevant")

                                if section.get("topic_buckets"):
                                    for bucket in section.get("topic_buckets", []):
                                        lines.append("- " + str(bucket.get("section", "")) + ": " + str(bucket.get("matching_source_posts", 0)) + " relevant posts")

                                if section.get("clusters"):
                                    for cluster_index, cluster in enumerate(section.get("clusters", [])[:15]):
                                        lines.append("Text cluster " + str(cluster_index + 1) + " | posts: " + str(cluster.get("repost_count", 0)) + " | distinct source groups: " + str(cluster.get("distinct_source_count", 0)) + " | " + str(cluster.get("repetition_type", "")))
                                        lines.append("Sources: " + ", ".join(cluster.get("source_names", [])[:8]))

                                if section.get("watch_indicators"):
                                    for indicator in section.get("watch_indicators", []):
                                        lines.append("- " + str(indicator))

                                if section.get("source_records"):
                                    lines.append("Selected source records:")
                                    record_limit = 8
                                    if heading.startswith("2."):
                                        record_limit = 20
                                    elif heading.startswith("3."):
                                        record_limit = 5
                                    elif heading.startswith("4."):
                                        record_limit = 10
                                    elif heading.startswith("5."):
                                        record_limit = 15
                                    for index, item in enumerate(section.get("source_records", [])[:record_limit]):

                                        body = " ".join(str(item.get("text", "")).split())[:320]

                                    for item in raw_candidates:

                                        elif unrest_hits:

                                            bucket = "security_action_lead"
                                        elif threat_hits:

                                            lines.append(str(index + 1) + ". " + str(publisher) + " | " + str(item.get("published_at") or "time unknown") + " | repeated same-source copies: " + str(entry["repeat_count"]))
