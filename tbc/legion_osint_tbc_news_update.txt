    try:

        if response.status_code != 200:

            summary = data.get("summary", {})
            bd = []
            global_items = []
            for post in posts:
                if "bangladesh" in str(post.get("priority_region", "")).lower() and len(bd) < 5:
                    bd.append(post)
                elif len(global_items) < 5:
                    global_items.append(post)
                if len(bd) >= 5 and len(global_items) >= 5:
                    break
            lines = ["LEGION-OSINT PUBLIC-SOURCE DIGEST — " + str(summary.get("report_date", "unknown")), "DRAFT SNAPSHOT ONLY. Not a verified incident report.             for post in bd:
                lines.append("- " + str(post.get("publisher_name") or post.get("channel_username", "unknown")) + " (" + str(post.get("platform", "Telegram")) + "): " + " ".join(str(post.get("text", "")).split())[:180] + " | " + str(post.get("source_url", "")))
            lines.append("\nOTHER REGIONAL SOURCE LEADS")
            for post in global_items:
                lines.append("- " + " ".join(str(post.get("text", "")).split())[:180] + " | " + str(post.get("source_url", "")))
