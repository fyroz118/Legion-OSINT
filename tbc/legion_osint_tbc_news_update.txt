    if not raw_query or raw_query == "none":
        bot.sendMessage("Usage: /search <words>\nExamples: /search Bangladesh protest | /search Hindustan | /search independent_media")
    else:
        try:
            query = raw_query.strip('"').strip("'")
            terms = [term for term in query.split() if term]
            response = HTTP.get(FEED_URL)
            if response.status_code != 200:
                bot.sendMessage("The public collection feed is temporarily unavailable. Try again later.")
            else:
                data = response.json()
                posts = data.get("posts", [])
                matches = []
                for post in posts:
                    fields = [
                        post.get("text", ""), post.get("channel_username", ""),
                        post.get("publisher_name", ""), post.get("platform", ""),
                        post.get("source_class", ""), post.get("source_type", ""),
                        post.get("source_affiliation", ""), post.get("priority_region", ""),
                        post.get("language", "")
                    ]
                    haystack = " ".join(str(value or "") for value in fields).lower()
                    if all(term in haystack for term in terms):
                        matches.append(post)
                    if len(matches) >= 8:
                        break
                if not matches:
                    bot.sendMessage("No matching posts in the latest collected snapshot. This searches collected records only, not all Telegram posts.")
                else:
                    lines = ["SEARCH RESULTS — UNVERIFIED SOURCE LEADS", "All search terms matched; results are from the latest collected snapshot."]

                        lines.append("\n" + str(post.get("publisher_name") or post.get("channel_username", "")) + " | " + str(post.get("platform", "Telegram")) + " | " + str(post.get("published_at") or "unknown") + "\n" + " ".join(str(post.get("text", "")).split())[:230] + "\n" + str(post.get("source_url", "")))

Command: date
---
if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    try:
        parts = str(params or "").strip().split()
        if len(parts) > 2:
            bot.sendMessage("Usage: /date YYYY-MM-DD [page]\\nExample: /date 2026-11-19 1")
        else:
            raw_date = ""
            if parts:
                raw_date = parts[0]
            else:
                latest_response = HTTP.get(FEED_URL)
                if latest_response.status_code != 200:
                    bot.sendMessage("Could not determine the latest reporting date. Try /date YYYY-MM-DD.")
                    raw_date = ""
                else:
                    latest_data = latest_response.json()
                    raw_date = str(latest_data.get("summary", {}).get("report_date") or "")
            if raw_date:
                valid_date = len(raw_date) == 10 and raw_date[4] == "-" and raw_date[7] == "-" and (raw_date[:4] + raw_date[5:7] + raw_date[8:]).isdigit()
                if valid_date:
                    year = int(raw_date[:4])
                    month = int(raw_date[5:7])
                    day = int(raw_date[8:10])
                    leap = year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)
                    month_days = [31, 29 if leap else 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
                    valid_date = year >= 1 and month >= 1 and month <= 12 and day >= 1 and day <= month_days[month - 1]
                if not valid_date:
                    bot.sendMessage("Invalid date. Use YYYY-MM-DD, for example /date 2026-11-19.")
                else:
                    page = 1
                    if len(parts) == 2:
                        if not parts[1].isdigit() or int(parts[1]) < 1:
                            bot.sendMessage("Invalid page. Use a positive page number.")
                            page = 0
                        else:
                            page = int(parts[1])
                    if page > 0:
                        page_url = "https://raw.githubusercontent.com/fyroz118/Legion-OSINT/main/docs/feed/archive/" + raw_date + "/page-" + str(page) + ".json"
                        page_response = HTTP.get(page_url)
                        if page_response.status_code == 200:
                            page_data = page_response.json()
                            page_posts = page_data.get("posts", [])
                            page_summary = page_data.get("summary", {})
                            total_pages = int(page_summary.get("total_pages") or 1)
                            total_records = int(page_summary.get("total_records") or len(page_posts))
                            if not page_posts:
                                bot.sendMessage("This archive page contains no records.")
                            else:
                                lines = ["LEGION-OSINT — DAILY NEWS: " + raw_date,
                                         "Page " + str(page) + "/" + str(total_pages) + " | Total records: " + str(total_records),
                                         "Public Telegram + RSS/Atom source leads. UNVERIFIED; reposts are not independent corroboration."]
                                for post in page_posts:
                                    publisher = str(post.get("publisher_name") or post.get("channel_username") or "Unknown source")
                                    platform = str(post.get("platform") or "Telegram")
                                    stamp = str(post.get("published_at") or "time unknown")
                                    body = " ".join(str(post.get("text") or "").split())[:190]
                                    url = str(post.get("source_url") or "")
                                    lines.append("\\n" + stamp + " | " + publisher + " | " + platform + "\\n" + body + "\\n" + url)
                                message_text = "\\n".join(lines)
                                bot.sendMessage(message_text[:3900], disable_web_page_preview=True)
                                if page < total_pages:
                                    bot.sendMessage("Next page: /date " + raw_date + " " + str(page + 1))
                        elif page_response.status_code == 404:
                            bot.sendMessage("Compact archive pages are not published for " + raw_date + " yet. They will appear after the next successful collection run. The old full archive is intentionally not downloaded because it can exceed TBC response limits.")
                        else:
                            bot.sendMessage("Archive page request failed (HTTP " + str(page_response.status_code) + "). Try again later.")
    except Exception as err:
        bot.sendMessage("Date command error: " + str(err)[:220] + ". Check TBC Error Logs.")
else:
    bot.sendMessage("Access denied. Legion-OSINT is restricted to its owner in private chat.")
