            terms = [term for term in query.split() if term]
            response = HTTP.get(FEED_URL)
            if response.status_code != 200:
                bot.sendMessage("The public collection feed is temporarily unavailable. Try again later.")
            else:
                data = response.json()
                posts = data.get("posts", [])
                matches = []
                for post in posts:
                    fields = [
                        post.get("text", ""), post.get("channel_username", ""),
                        post.get("publisher_name", ""), post.get("platform", ""),
                        post.get("source_class", ""), post.get("source_type", ""),
                        post.get("source_affiliation", ""), post.get("priority_region", ""),
                        post.get("language", "")
                    ]
                    haystack = " ".join(str(value or "") for value in fields).lower()
                    if all(term in haystack for term in terms):
                        matches.append(post)
                    if len(matches) >= 8:
                        break
                if not matches:
                    bot.sendMessage("No matching posts in the latest collected snapshot. This searches collected records only, not all Telegram posts.")
                else:
                    lines = ["SEARCH RESULTS — UNVERIFIED SOURCE LEADS", "All search terms matched; results are from the latest collected snapshot."]

if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    try:
        parts = str(params or "").strip().split()
        if len(parts) > 2:
            bot.sendMessage("Usage: /date YYYY-MM-DD [page]\nExample: /date 2026-11-18 2")
        else:
            feed_url = FEED_URL
            feed_response = HTTP.get(feed_url)
            if feed_response.status_code != 200:
                bot.sendMessage("The collection feed is unavailable. Try again later.")
            else:
                latest = feed_response.json()
                raw_date = parts[0] if parts else str(latest.get("summary", {}).get("report_date") or "")
                page = int(parts[1]) if len(parts) == 2 and parts[1].isdigit() else (1 if len(parts) == 1 else 0)
                valid_date = len(raw_date) == 10 and raw_date[4] == "-" and raw_date[7] == "-" and (raw_date[:4] + raw_date[5:7] + raw_date[8:]).isdigit()
                if valid_date:
                    year = int(raw_date[:4])
                    month = int(raw_date[5:7])
                    day = int(raw_date[8:10])
                    leap = year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)
                    month_days = [31, 29 if leap else 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
                    valid_date = year >= 1 and month >= 1 and month <= 12 and day >= 1 and day <= month_days[month - 1]
                if not valid_date:
                    bot.sendMessage("Usage: /date YYYY-MM-DD [page]\nExample: /date 2026-11-18 2")
                elif page < 1:
                    bot.sendMessage("Invalid page. Use a positive page number, for example /date 2026-11-18 2.")
                else:
                    archive_url = "https://raw.githubusercontent.com/fyroz118/Legion-OSINT/main/docs/feed/archive/" + raw_date + ".json"
                    response = HTTP.get(archive_url)
                    is_fallback = False
                    if response.status_code == 200:
                        data = response.json()
                        posts = data.get("posts", [])
                        summary = data.get("summary", {})
                        timezone_name = str(summary.get("archive_timezone") or summary.get("timezone") or "Asia/Dhaka")
                    elif response.status_code == 404 and raw_date == str(latest.get("summary", {}).get("report_date") or ""):
                        is_fallback = True
                        data = latest
                        posts = data.get("posts", [])
                        summary = data.get("summary", {})
                        timezone_name = str(summary.get("timezone") or "Asia/Dhaka")
                        bot.sendMessage("Daily archive not published yet; showing matching records from the latest snapshot.")
                    elif response.status_code == 404:
                        posts = []
                        timezone_name = "Asia/Dhaka"
                    else:
                        bot.sendMessage("The daily archive is temporarily unavailable. Try again later.")
                        posts = None
                    if posts is not None:
                        if is_fallback:
                            posts = [p for p in posts if str(p.get("published_at") or "").startswith(raw_date)]
                        posts.sort(key=lambda item: str(item.get("published_at") or ""))
                        if not posts:
                            bot.sendMessage("No archived records found for " + raw_date + ". Older dates cannot be reconstructed if no archive was created on that date.")
                        else:
                            page_size = 20
                            total_pages = (len(posts) + page_size - 1) // page_size
                            if page > total_pages:
                                bot.sendMessage("Page " + str(page) + " does not exist. Available pages: 1-" + str(total_pages) + ".")
                            else:
                                selected = posts[(page - 1) * page_size:page * page_size]
                                lines = ["LEGION-OSINT — DAILY NEWS: " + raw_date, "Timezone: " + timezone_name + " | Total records: " + str(len(posts)), "Page " + str(page) + "/" + str(total_pages) + " | Records: " + str(len(selected)), "Telegram + RSS/Atom public-source leads. UNVERIFIED; multiple posts may refer to the same event."]
                                for post in selected:
                                    publisher = str(post.get("publisher_name") or post.get("channel_username") or "Unknown source")
                                    platform = str(post.get("platform") or "Telegram")
                                    stamp = str(post.get("published_at") or "time unknown")
                                    body = " ".join(str(post.get("text") or "").split())[:190]
                                    url = str(post.get("source_url") or "")
                                    lines.append("\\n" + stamp + " | " + publisher + " | " + platform + "\\n" + body + "\\n" + url)
                                current = "\\n".join(lines[:4])
                                chunks = []
                                for item in lines[4:]:
                                    if len(current) + len(item) + 1 > 3800:
                                        chunks.append(current)
                                        current = "CONTINUED — " + raw_date + " | PAGE " + str(page)
                                    current += "\\n" + item
                                if current:
                                    chunks.append(current)
                                for chunk in chunks[:6]:
                                    bot.sendMessage(chunk, disable_web_page_preview=True)
                                if page < total_pages:
                                    bot.sendMessage("More records: /date " + raw_date + " " + str(page + 1))
    except Exception:
        bot.sendMessage("Date retrieval failed. Check TBC Error Logs; confirm HTTP access to raw.githubusercontent.com.")
else:
    bot.sendMessage("Access denied. Legion-OSINT is restricted to its owner in private chat.")

    try:
        response = HTTP.get("https://raw.githubusercontent.com/fyroz118/Legion-OSINT/main/discovery/seed_candidates.json")
        if response.status_code != 200:
            bot.sendMessage("Candidate registry unavailable.")
        else:
            data = response.json()
            items = data.get("candidates", [])
            pending = [x for x in items if x.get("review_status") == "UNREVIEWED_CANDIDATE" and not x.get("enabled_for_collection", False)]
            lines = ["LEGION-OSINT — LEGACY / EXCEPTION REVIEW QUEUE", "Pending candidates: " + str(len(pending)), "New accessible discoveries are auto-added. Manual override: /addchannel USERNAME | /rejectchannel USERNAME"]
            for x in pending[:15]:
                lines.append("\n@" + str(x.get("username", "")) + " | " + str(x.get("region", "unknown")) + " | " + str(x.get("candidate_category", "unknown")) + "\n" + str(x.get("seed_source_url", "")))
            bot.sendMessage("\n".join(lines)[:3900], disable_web_page_preview=True)
    except Exception:
        bot.sendMessage("Candidate registry unavailable.")
else:
    bot.sendMessage("Access denied.")

if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    username = str(params or "").strip().lstrip("@")
    if not username or not username.replace("_", "").isalnum():
        bot.sendMessage("Usage: /addchannel USERNAME\nThis approves a candidate already in the discovery registry.")
    elif not GITHUB_TOKEN:
        bot.sendMessage("GitHub decision token is not configured. Add GITHUB_TOKEN to the TBC environment before using this command.")
    else:
        try:
            url = "https://api.github.com/repos/fyroz118/Legion-OSINT/dispatches"
            payload = {"event_type": "channel_approve", "client_payload": {"username": username, "decision": "approve"}}
            response = HTTP.post(url, headers={"Accept": "application/vnd.github+json", "Authorization": "Bearer " + GITHUB_TOKEN, "X-GitHub-Api-Version": "2022-11-28"}, json=payload)
            if response.status_code == 204:
                bot.sendMessage("Approval request submitted for @" + username + ". GitHub Actions will add it if it exists in the candidate registry. Check /discovery_status and /coverage after the workflow completes.")
            else:
                bot.sendMessage("GitHub rejected the approval request. Check token permissions and TBC Error Logs. HTTP " + str(response.status_code))
        except Exception:
            bot.sendMessage("Could not submit approval. Check GITHUB_TOKEN permissions and TBC HTTP support.")
else:
    bot.sendMessage("Access denied.")
