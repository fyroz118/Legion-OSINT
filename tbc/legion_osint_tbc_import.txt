---
if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    ids = Bot.getData("legion_incident_ids") or []
    approved = []
    draft_count = 0
    rejected_count = 0
    for incident_id in ids:
        item = Bot.getData("legion_incident_" + str(incident_id))
        if item:
            status = item.get("inclusion_status", "DRAFT")
            if status == "APPROVED":
                approved.append(item)
            elif status == "REJECTED":
                rejected_count += 1
            else:
                draft_count += 1
    lines = ["LEGION-OSINT INCIDENT REPORT", "Inclusion: analyst-approved records only. Approval does not mean verification.", "Approved: " + str(len(approved)) + " | Draft: " + str(draft_count) + " | Rejected: " + str(rejected_count)]
    if not approved:
        lines.append("No approved incident records are currently available. This does not mean no incidents occurred.")
    else:
        for item in approved[-20:]:
            lines.append("\n" + str(item.get("incident_id", "")) + " | " + str(item.get("country", "Unknown")) + " | " + str(item.get("category", "Unknown")))
            lines.append("Verification: " + str(item.get("verification_status", "UNVERIFIED")) + " | Confidence: " + str(item.get("confidence", "Low")) + " | Risk: " + str(item.get("risk_level", "Low")))
            lines.append(str(item.get("description", ""))[:300])
            for source in item.get("sources", []):
                lines.append("[" + str(source.get("source_role", "lead")) + "] " + str(source.get("source_class", "unknown")) + " | " + str(source.get("source_url", "")))
    lines.append("Publication time is not incident time. Reposts and same-publisher posts are not independent corroboration.")
    bot.sendMessage("\n".join(lines)[:3900], disable_web_page_preview=True)

Command: reviewlead
---
if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    parts = [x.strip() for x in str(params or "").split("|", 3)]
    if len(parts) < 4 or not parts[3]:
        bot.sendMessage("Usage: /reviewlead POST_ID | reviewed|rejected|duplicate | source_class\nAllowed classes: unofficial, independent_media, official, institutional, advocacy, unknown")
    else:
        post_id = parts[0]
        decision = parts[1].lower()
        source_class = parts[2].lower()
        rationale = parts[3]
        allowed_classes = ["unofficial", "independent_media", "official", "institutional", "advocacy", "unknown"]
        if decision not in ["reviewed", "rejected", "duplicate"] or source_class not in allowed_classes:
            bot.sendMessage("Invalid decision or source class.")
        else:
            try:
                response = HTTP.get(FEED_URL)
                if response.status_code != 200:
                    bot.sendMessage("The public collection feed is temporarily unavailable. Try again later.")
                else:
                    data = response.json()
                    posts = data.get("posts", [])
                    summary = data.get("summary", {})
                    found = False
                    for post in posts:
                        if str(post.get("id", "")) == post_id:
                            found = True
                            break
                    if not found:
                        bot.sendMessage("Post ID not found in the current feed. It may have expired from the snapshot.")
                    else:
                        Bot.saveData("legion_review_" + post_id, {"status": decision.upper(), "source_class": source_class, "reviewed_by": str(u), "rationale": rationale, "feed_snapshot_at": str(data.get("generated_at", ""))})
                        bot.sendMessage("Saved analyst decision: " + decision.upper() + ". This does not verify the underlying claim.")
            except Exception:
                bot.sendMessage("Could not save review decision.")
else:
    bot.sendMessage("Access denied. Legion-OSINT is restricted to its owner in private chat.")

Command: incidentdraft
---
if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    parts = [x.strip() for x in str(params or "").split("|", 4)]
    if len(parts) < 5:
        bot.sendMessage("Usage: /incidentdraft POST_ID | INCIDENT_ID | COUNTRY | CATEGORY | DESCRIPTION")
    else:
        post_id, incident_id, country, category, description = parts
        existing = Bot.getData("legion_incident_" + incident_id)
        if existing:
            bot.sendMessage("That incident ID already exists. Use a new ID; existing analyst records will not be overwritten.")
        elif not incident_id or len(incident_id) > 64 or " " in incident_id:
            bot.sendMessage("Use a unique incident ID of up to 64 characters without spaces.")
        else:
            review = Bot.getData("legion_review_" + post_id)
            if not review or review.get("status") != "REVIEWED" or review.get("source_class") == "unknown":
                bot.sendMessage("Only a REVIEWED post with an analyst-assigned source class can seed a draft.")
            else:
                try:
                    response = HTTP.get(FEED_URL)
                    if response.status_code != 200:
                        bot.sendMessage("The public collection feed is temporarily unavailable. Try again later.")
                    else:
                        data = response.json()
                        posts = data.get("posts", [])
                        source = None
                        for post in posts:
                            if str(post.get("id", "")) == post_id:
                                source = post
                                break
                        if not source:
                            bot.sendMessage("Source post not found in current feed.")
                        else:
                            incident = {"incident_id": incident_id, "country": country, "category": category, "description": description, "verification_status": "UNVERIFIED", "confidence": "Low", "risk_level": "Low", "inclusion_status": "DRAFT", "approval_rationale": "", "sources": [{"post_id": post_id, "source_url": source.get("source_url"), "source_name": source.get("channel_username"), "source_class": review.get("source_class"), "source_role": "lead", "independence_group": source.get("channel_username")}]}
                            Bot.saveData("legion_incident_" + incident_id, incident)
                            ids = Bot.getData("legion_incident_ids") or []
                            if incident_id not in ids:
                                ids.append(incident_id)
                            Bot.saveData("legion_incident_ids", ids)
                            bot.sendMessage("Incident draft created. Verification remains UNVERIFIED; confidence and risk remain Low. No incident date or precise location was inferred.")
                except Exception:
                    bot.sendMessage("Could not create incident draft.")
else:
    bot.sendMessage("Access denied. Legion-OSINT is restricted to its owner in private chat.")

Command: approveincident
---
if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    parts = [x.strip() for x in str(params or "").split("|", 2)]
    if len(parts) < 3 or parts[1].lower() not in ["approved", "rejected"]:
        bot.sendMessage("Usage: /approveincident INCIDENT_ID | approved|rejected | rationale")
    else:
        incident_id, decision, rationale = parts
        item = Bot.getData("legion_incident_" + incident_id)
        if not item:
            bot.sendMessage("Incident not found.")
        elif not rationale:
            bot.sendMessage("A rationale is required.")
        elif decision.lower() == "approved" and not item.get("sources"):
            bot.sendMessage("Cannot approve for report inclusion without an attached source.")
        else:
            item["inclusion_status"] = decision.upper()
            item["approval_rationale"] = rationale
            Bot.saveData("legion_incident_" + incident_id, item)
            bot.sendMessage("Inclusion decision saved: " + decision.upper() + ". This is not a verification decision; verification remains " + str(item.get("verification_status", "UNVERIFIED")) + ".")
else:
    bot.sendMessage("Access denied. Legion-OSINT is restricted to its owner in private chat.")

Command: linklead
---
if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    parts = [x.strip() for x in str(params or "").split("|", 2)]
    if len(parts) < 2:
        bot.sendMessage("Usage: /linklead POST_ID | INCIDENT_ID | lead|corroboration|contradiction")
    else:
        post_id, incident_id = parts[0], parts[1]
        role = parts[2].lower() if len(parts) > 2 else "lead"
        item = Bot.getData("legion_incident_" + incident_id)
        review = Bot.getData("legion_review_" + post_id)
        if not item or not review or review.get("status") != "REVIEWED" or role not in ["lead", "corroboration", "contradiction"]:
            bot.sendMessage("Incident, reviewed source, or role is invalid.")
        else:
            try:
                response = HTTP.get(FEED_URL)
                if response.status_code != 200:
                    bot.sendMessage("The public collection feed is temporarily unavailable. Try again later.")
                else:
                    data = response.json()
                    posts = data.get("posts", [])
                    summary = data.get("summary", {})
                    source = None
                    for post in posts:
                        if str(post.get("id", "")) == post_id:
                            source = post
                            break
                    if not source:
                        bot.sendMessage("Source post not found in current feed.")
                    else:
                        exists = False
                        for linked in item.get("sources", []):
                            if linked.get("post_id") == post_id:
                                exists = True
                        if not exists:
                            item["sources"].append({"post_id": post_id, "source_url": source.get("source_url"), "source_name": source.get("channel_username"), "source_class": review.get("source_class"), "source_role": role, "independence_group": source.get("channel_username")})
                            Bot.saveData("legion_incident_" + incident_id, item)
                        bot.sendMessage("Source link saved as " + role + ". Independence must be assessed; multiple posts from the same publisher are not independent corroboration.")
            except Exception:
                bot.sendMessage("Could not link source.")
else:
    bot.sendMessage("Access denied. Legion-OSINT is restricted to its owner in private chat.")

if str(u) in [str(x) for x in OWNER_IDS] and message.chat.type == "private":
    try:
        response = HTTP.get(FEED_URL)
        if response.status_code != 200:
            bot.sendMessage("The public collection feed is temporarily unavailable. Try again later.")
        else:
            data = response.json()
            posts = data.get("posts", [])
            summary = data.get("summary", {})
            bd = []
            global_items = []
            for post in posts:
                if "bangladesh" in str(post.get("priority_region", "")).lower() and len(bd) < 5:
                    bd.append(post)
                elif len(global_items) < 5:
                    global_items.append(post)
                if len(bd) >= 5 and len(global_items) >= 5:
                    break
            lines = ["LEGION-OSINT PUBLIC-SOURCE DIGEST — " + str(summary.get("report_date", "unknown")), "DRAFT SNAPSHOT ONLY. Not a verified incident report. Empty sections do not indicate absence of events.", "Feed generated: " + str(summary.get("generated_at", "unknown")), "Configured channels: " + str(summary.get("configured_channels", 0)) + " | Collection errors: " + str(summary.get("collection_errors", 0)), "\nBANGLADESH SOURCE LEADS"]
            for post in bd:
                lines.append("- " + " ".join(str(post.get("text", "")).split())[:180] + " | " + str(post.get("source_url", "")))
            lines.append("\nOTHER REGIONAL SOURCE LEADS")
            for post in global_items:
                lines.append("- " + " ".join(str(post.get("text", "")).split())[:180] + " | " + str(post.get("source_url", "")))
            lines.append("\nAll posts remain UNVERIFIED. Publication time is not necessarily incident time; duplicate reposts are not independent corroboration.")
            bot.sendMessage("\n".join(lines)[:3900], disable_web_page_preview=True)
    except Exception:
        bot.sendMessage("Could not build the evidence digest from the current feed.")
